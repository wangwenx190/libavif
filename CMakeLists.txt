cmake_minimum_required(VERSION 3.25)
project(libavif VERSION 1.4.2 LANGUAGES C CXX)
include(GNUInstallDirs)

option(AVIF_BUILD_STATIC "Build the static library" ON)
option(AVIF_BUILD_SHARED "Build the shared library" ON)
if(TARGET aom)
    set(aom_library aom)
else()
    find_path(aom_include aom/aom.h REQUIRED)
    find_library(aom_binary NAMES aom aom_static REQUIRED)
    add_library(avif_aom UNKNOWN IMPORTED)
    set_target_properties(avif_aom PROPERTIES IMPORTED_LOCATION "${aom_binary}"
        INTERFACE_INCLUDE_DIRECTORIES "${aom_include}")
    find_package(Threads REQUIRED)
    target_link_libraries(avif_aom INTERFACE Threads::Threads)
    if(UNIX)
        target_link_libraries(avif_aom INTERFACE m)
    endif()
    set(aom_library avif_aom)
endif()
include(cmake/Libyuv.cmake)

file(GLOB avif_sources CONFIGURE_DEPENDS src/*.c)
list(FILTER avif_sources EXCLUDE REGEX "/codec_.*[.]c$")
list(APPEND avif_sources src/codec_aom.c)
function(add_avif target type)
    add_library(${target} ${type} ${avif_sources})
    target_compile_features(${target} PRIVATE c_std_99)
    target_include_directories(${target} PUBLIC
        "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
        "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>" PRIVATE third_party/libyuv/include)
    target_compile_definitions(${target} PRIVATE AVIF_CODEC_AOM=1
        AVIF_CODEC_AOM_ENCODE=1 AVIF_CODEC_AOM_DECODE=1 AVIF_LIBYUV_ENABLED=1)
    target_link_libraries(${target} PRIVATE ${aom_library} avif_yuv)
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(AVIF_INSTALL)
        install(TARGETS ${target})
    endif()
endfunction()
option(AVIF_INSTALL "Install libavif" ON)
if(AVIF_BUILD_STATIC)
    add_avif(avif STATIC)
    if(WIN32)
        set_target_properties(avif PROPERTIES OUTPUT_NAME avif_static)
    endif()
endif()
if(AVIF_BUILD_SHARED)
    add_avif(avif_shared SHARED)
    target_compile_definitions(avif_shared PRIVATE AVIF_BUILDING_SHARED_LIBS PUBLIC AVIF_DLL)
    set_target_properties(avif_shared PROPERTIES OUTPUT_NAME avif
        VERSION ${PROJECT_VERSION} SOVERSION 16)
endif()
if(AVIF_INSTALL)
    install(DIRECTORY include/avif DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
    install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/libavif")
    install(FILES third_party/libyuv/LICENSE third_party/libyuv/PATENTS third_party/libyuv/AUTHORS
        DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/libavif/libyuv")
endif()

if(WIN32 AND TARGET avif_shared)
    enable_language(RC)
    configure_file(cmake/LibraryVersion.rc.in LibraryVersion.rc @ONLY)
    target_sources(avif_shared PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/LibraryVersion.rc")
endif()
